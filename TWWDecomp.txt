local function initClothing(p47, p48) -- Line: 8573
    -- upvalues: u2 (copy), initClothing (copy), Global (copy)
    u2.Clothing[p47] = u2.Clothing[p47] or {};

    for _, child in p48:GetChildren() do
        if child:IsA("Folder") then
            initClothing(p47, child);
        else
            local Name = child.Name;

            if p47 ~= "Holsters" or not Name:match("Left$") then
                local v49;

                if Global.IsServer and child:IsA("Accessory") then
                    child:SetAttribute("ItemType", p47);
                    v49 = child;

                    for _, descendant in child:GetDescendants() do
                        if descendant:IsA("BasePart") then
                            descendant.CanCollide = false;
                            descendant.Massless = true;
                        end;
                    end;

                    local Handle = v49.Handle;
                    local v50 = not Handle:FindFirstChild("OriginalSize") and Vector3.new(1, 1, 1) or Handle.Size / Handle.OriginalSize.Value;

                    for _, descendant in v49:GetDescendants() do
                        if descendant:IsA("BasePart") then
                            descendant:SetAttribute("OriginalSize", descendant.Size / v50);
                        elseif descendant:IsA("Attachment") then
                            descendant:SetAttribute("OriginalPosition", descendant.Position / v50);
                        elseif descendant:IsA("SpecialMesh") then
                            descendant:SetAttribute("OriginalOffset", descendant.Offset / v50);
                            descendant:SetAttribute("OriginalScale", descendant.Scale / v50);
                        elseif descendant:IsA("JointInstance") then
                            descendant:SetAttribute("OriginalC0", descendant.C0.Position / v50);
                            descendant:SetAttribute("OriginalC1", descendant.C1.Position / v50);
                        end;
                    end;
                else
                    v49 = child;
                end;

                u2.Clothing[p47][Name] = {
                    name = Name,
                    obj = v49,
                    OwnedByDefault = v49:GetAttribute("OwnedByDefault") == true,
                    HidePreview = v49:GetAttribute("HidePreview") == true
                };
            end;
        end;
    end;
end;

local Clothing = Global.Resources.Clothing;

end;

for _, child in Global.Resources.Clothing:GetChildren() do
